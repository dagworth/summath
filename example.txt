x*=4                      {x:48} (x)

y=x                      {y:13} (x)
y+x                      {} (x y)

def hi(w p)           {hi(1,2) w:12, p:2 scope down} (x p z)
     z  = w*4          {z:48}   (w)
     w++                 {w:13}  (w)
     x=12                {x:12}    ()
     return x+p+z    {}       (x p z)
end                      {scope up}
  
hi(12,2)            {x:12}   (hi)
 
for 3 do                {scope down}   () looped 3

     x/=2                {x:24, x:12, x:6}  (x)
end                      {scope up}
